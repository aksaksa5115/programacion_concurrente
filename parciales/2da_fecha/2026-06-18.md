Luego el profesor va corrigiendo los exámenes y enviando la nota de acuerdo con orden en que se van entregando. Cada alumno realiza el examen, lo entrega y espera a que el profesor le indique la nota. Nota: maximizar la concurrencia; no realizar busy waiting; todos los procesos deben terminar.

chan llegue(int);

chan entrega(text, int);
chan notas[1..E](int);

Process Estudiante[id: 1..E] {

    send llegue(id);

    examen = resolverExamen(enun);

Process Profesor {

2) Resolver con PASAJE DE MENSAJES SINCRÓNICOS (PMS) el siguiente problema. En un consultorio hay un médico para atender a 15 pacientes de acuerdo con el orden de llegada. Cada paciente al llegar espera a que el médico lo atiende y le indique su tratamiento. Nota: existe la función atender() que simula que el médico está atendiendo a un paciente; todos los procesos deben terminar.


Process Paciente[id: 1..15] {
    text tratamiento;
    Coordinador ! llegue(id);
    Medico ? receta(tratamiento);
}

Process Coordinador {
    cola fila;
    int idP, cant = 0;
    do (cant < 15); Paciente[*] ? llegue(idP) ->
            push(fila, idP);
            cant = cant + 1;
       (not empty(fila)); Medico ? libre() ->
            pop(fila, idP);
            Medico ! siguiente(idP);
    od
}

Process Medico {
    int idP;
    text tratamiento;
    for (i = 1..15) {
        Coordinador ! libre();
        Coordinador ? siguiente(idP);
        tratamiento = atender(idP);
        Paciente[idP] ! receta(tratamiento);
    }
}

3. Resolver con ADA el siguiente problema. En una competencia de programadores hay P participantes y UN coordinador. El coordinador entrega el problema a resolver a los participantes y recibe las resoluciones para corregir. Cada participante debe conocer el resultado de su trabajo y el orden en que entregó su resolución. Nota: maximizar la concurrencia.

procedure Competencia is

   TASK TYPE Programador IS
      ENTRY Problema (Enunciado: IN texto);
   END Programador;

   arrProgramadores: ARRAY (1..P) OF Programador;

   TASK Coordinador IS
   ENTRY PedirProblema (Enunciado: OUT texto);
   ENTRY Solucion (Trabajo: IN texto; Nota: OUT integer; Orden: OUT integer);
    END Coordinador;

    TASK BODY Coordinador IS
    enunciado: texto;
    contador: integer := 0;
    BEGIN
    enunciado := generarProblema();

    FOR i IN 1..2*P LOOP                 -- P pedidos de problema + P soluciones
        SELECT
            ACCEPT PedirProblema (Enunciado: OUT texto) DO
                Enunciado := enunciado;
            END PedirProblema;
        OR
            ACCEPT Solucion (Trabajo: IN texto; Nota: OUT integer; Orden: OUT integer) DO
                contador := contador + 1;
                Orden := contador;
                Nota := corregir(Trabajo);
            END Solucion;
        END SELECT;
    END LOOP;
    END Coordinador;

    TASK BODY Programador IS
    miProblema, miSolucion: texto;
    miNota, miOrden: integer;
    BEGIN
    Coordinador.PedirProblema(miProblema);
    miSolucion := resolverProblema(miProblema);
    Coordinador.Solucion(miSolucion, miNota, miOrden);
    END Programador;

BEGIN
   null;
END Competencia;
